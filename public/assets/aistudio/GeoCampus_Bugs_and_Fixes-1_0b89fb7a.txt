BUG 1 (CRITICAL): Passwords hardcoded in plain text
---------------------------------------------------------------------

  Problem:
    - The `users` array in the JavaScript shows every password to anyone who
      views the page source.

  Solution:
    - Remove the `users` array from the frontend completely.
    - Store users in the database with hashed passwords.
    - Login is a POST request to the server.

  Server code (auth.js):

    const bcrypt = require('bcrypt');
    const jwt = require('jsonwebtoken');

    // Create a user (run once in a seed script)
    async function createUser(db, id, name, role, plainPassword) {
      const hash = await bcrypt.hash(plainPassword, 12);
      db.prepare('INSERT INTO users (id,name,role,password_hash) VALUES (?,?,?,?)')
        .run(id, name, role, hash);
    }

    // Login route
    app.post('/api/login', loginLimiter, async (req, res) => {
      const { id, password, role } = req.body;
      const user = db.prepare('SELECT * FROM users WHERE id=? AND role=?').get(id, role);
      if (!user || !(await bcrypt.compare(password, user.password_hash))) {
        return res.status(401).json({ error: 'Invalid credentials' });
      }
      const token = jwt.sign({ id: user.id, role: user.role, name: user.name },
                             process.env.JWT_SECRET, { expiresIn: '2h' });
      res.json({ token, user: { id: user.id, name: user.name, role: user.role } });
    });

  Frontend change (replace handleLogin):

    async function handleLogin() {
      const role = document.getElementById('loginRole').value;
      const id = document.getElementById('loginId').value.trim();

      const data = await res.json();
      sessionStorage.setItem('jwt', data.token);   // only the token, no password, no role trust
      currentUser = data.user;
      showDashboard();
    }

  Also remove the pre-filled values (value="CIT2026001" and value="student123")
  from the login inputs.

BUG 2 (CRITICAL): Login state and role stored in localStorage
---------------------------------------------------------------------

  Problem:
    - Anyone can run localStorage.setItem("currentUser", '{"role":"admin"}')
      and become admin.

  Solution:
    - The role comes ONLY from the signed JWT, and the server checks it on every request.
    - The frontend role is used only to decide which panels to show (cosmetic).
    - Real protection is on the server.

  Server middleware:

    function requireAuth(req, res, next) {
      const h = req.headers.authorization || '';
      try {
        req.user = jwt.verify(h.replace('Bearer ', ''), process.env.JWT_SECRET);
        next();
      } catch { res.status(401).json({ error: 'Unauthorized' }); }
    }

    function requireRole(...roles) {
      return (req, res, next) =>
        roles.includes(req.user.role) ? next() : res.status(403).json({ error: 'Forbidden' });
    }

BUG 3 (CRITICAL): Attendance, tokens and geofence validated in the browser
---------------------------------------------------------------------

  Problem:
    - performCheckIn() checks the token, distance and writes the log on the
      client. A student can edit attendanceLogs in DevTools or call the function
      with fake coordinates.

  Solution:
    - The student sends: token + latitude + longitude + accuracy.
    - The SERVER verifies token validity, expiry, duplicates, and distance.
    - The server writes the log.

  Server code (geo.js + route):

    function haversine(lat1, lon1, lat2, lon2) {
      const R = 6371000, toRad = d => d * Math.PI / 180;
      const dLat = toRad(lat2 - lat1), dLon = toRad(lon2 - lon1);
      const a = Math.sin(dLat/2)**2 +
                Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLon/2)**2;
      return 2 * R * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    }

    app.post('/api/checkin', requireAuth, requireRole('student'), (req, res) => {
      const { token, lat, lon, accuracy } = req.body;
      if (![lat, lon, accuracy].every(Number.isFinite)) {
        return res.status(400).json({ error: 'Bad input' });
      }

      const session = db.prepare(
        'SELECT * FROM sessions WHERE token=? AND expires_at > ?'
      ).get(String(token).toUpperCase(), Date.now());
      if (!session) return res.status(400).json({ error: 'Invalid or expired token' });

      if (accuracy > 50) return res.status(400).json({ error: 'GPS accuracy too low' });

      const cfg = db.prepare('SELECT * FROM config WHERE id=1').get();
      const dist = haversine(lat, lon, cfg.lat, cfg.lon);
      const inside = dist <= cfg.radius;     // see Bug 5 about not adding accuracy

      const already = db.prepare(
        "SELECT 1 FROM attendance WHERE user_id=? AND session_id=? AND status='Verified'"
      ).get(req.user.id, session.id);
      if (already) return res.status(409).json({ error: 'Already marked' });

      db.prepare(`INSERT INTO attendance
        (session_id,user_id,course,lat,lon,distance,accuracy,status,created_at)
        VALUES (?,?,?,?,?,?,?,?,?)`)
        .run(session.id, req.user.id, session.course, lat, lon,
             Math.round(dist), accuracy, inside ? 'Verified' : 'Rejected', Date.now());

      res.json({ status: inside ? 'Verified' : 'Rejected', distance: Math.round(dist) });
    });

  Frontend (replace performCheckIn): send fetch('/api/checkin', {...}) with the
  Authorization header. Delete the local attendanceLogs.unshift(...) logic.

BUG 4 (CRITICAL): Faculty token never reaches student devices
---------------------------------------------------------------------

  Problem:
    - activeSession is in localStorage, which only exists on the same browser.
      Students on their own phones will always see "No active session".

  Solution:
    - Faculty creates the session via the API, the server stores it.

  Server code:

    app.post('/api/sessions', requireAuth, requireRole('faculty'), (req, res) => {
      const course = String(req.body.course || '').toUpperCase().slice(0, 12);
      if (!/^[A-Z0-9]{2,12}$/.test(course)) return res.status(400).json({ error: 'Bad course' });
      const code = crypto.randomInt(100000, 999999);   // use crypto, not Math.random
      const token = `${course}-${code}`;
      const expires = Date.now() + 60_000;
      const info = db.prepare(
        'INSERT INTO sessions (token,course,faculty_id,expires_at) VALUES (?,?,?,?)'
      ).run(token, course, req.user.id, expires);
      res.json({ id: info.lastInsertRowid, token, expiresAt: expires });
    });

  Add: const crypto = require('crypto');

=====================================================================
TERMS EXPLAINED
=====================================================================

  - Backend: server code and database that users cannot edit from their browser.

  - JWT (JSON Web Token): a signed pass the server gives after login. The server
    checks its signature on every request, so it cannot be faked without the secret.

  - bcrypt / hashing: a one-way scrambling of passwords, so even the database owner
    cannot read the original.

  - RBAC (Role-Based Access Control): rules that decide what each role
    (student, faculty, admin) is allowed to do.

  - XSS (Cross-Site Scripting): attacker-written script running inside your page
    because user text was inserted as HTML.

  - SRI (Subresource Integrity): a hash that makes the browser reject a CDN file
    if it was changed.

  - CSP (Content-Security-Policy): a browser rule listing which scripts and
    sources the page may load.

  - Rate limiting: capping how many requests one user/IP can make, to stop
    password guessing.

  - Haversine formula: calculates the distance between two GPS points on a sphere.

  - Geofence: a virtual boundary, here a circle around the campus center.

  - TOTP: time-based one-time passwords (the 6-digit codes in authenticator apps).

  - Parameterized query: a database query where values are passed separately
    (the "?" placeholders), which prevents SQL injection.

  - SQL injection: attacker input that changes the meaning of a database query.

  - Toast: a small temporary notification message that does not block the page.
